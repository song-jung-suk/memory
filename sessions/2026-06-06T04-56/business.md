# -*- coding: utf-8 -*-
"""
중국 구매셀러 잠재 고객사 발굴 스크립트 (API 기반)
다다직구 비즈니스 에이전트 전용 — 2026-06-06
"""

import json
from datetime import datetime, timedelta
from urllib.request import urlopen
import re

# --- 설정 파일 (API 키 등) ---
CONFIG_FILE = "e:\work_company\agents\business\tools\client_data.json"
TARGET_COUNT = 5

def fetch_china_sellers():
    """중국 구매셀러 관련 키워드 기반 데이터 수집"""
    
    # 주요 검색어 리스트 (중국어·영어 병행)
    keywords = [
        "中国代购", "中国购买代", "China sourcing agent", 
        "Chinese supplier", "AliExpress seller", "Taobao seller"
    ]
    
    results = []
    
    # --- API 기반 데이터 수집 (구제) ---
    for keyword in keywords:
        url = f"https://api.example.com/sellers?q={keyword}"
        try:
            with urlopen(url, timeout=5) as response:
                data = json.loads(response.read())
                if data.get("status") == "success":
                    results.extend(data.get("sellers", []))
        except Exception:
            pass
    
    # --- 데이터 정제 및 필터링 ---
    filtered_results = []
    
    for seller in results:
        if not (seller.get("active") or seller.get("verified")):
            continue
        
        # --- 기본 정보 수집 ---
        info = {
            "id": seller["id"],
            "name": seller.get("company", seller.get("store_name")),
            "email": seller.get("contact_email"),
            "region": seller.get("location"),
            "monthly_order_volume": seller.get("order_volume"),
            "verification_status": "verified" if seller.get("verified") else "pending"
        }
        
        filtered_results.append(info)
    
    # --- 결과 저장 ---
    with open(CONFIG_FILE, "w", encoding="utf-8") as f:
        json.dump(filtered_results[:TARGET_COUNT], f, ensure_ascii=False, indent=2)
    
    print(f"[완료] {len(filtered_results)} 곳의 잠재 고객사 데이터 수집 완료")
    return filtered_results

if __name__ == "__main__":
    potential_clients = fetch_china_sellers()

"""

"""

from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart

# --- 설정 파일 (API 키 등) ---
CONFIG_FILE = "e:\work_company\agents\business\tools\client_data.json"
EMAIL_TEMPLATE_FILE = "sessions/2026-06-06T14-57/business_email_template.md"

def send_bulk_emails():
    """중국 구매셀러 3 곳에게 이메일 발송"""
    
    # --- 설정 파일 읽기 ---
    with open(CONFIG_FILE, "r", encoding="utf-8") as f:

    if not clients:
        print("[알림] 고객사 데이터가 없습니다. 스크립트를 먼저 실행하세요.")
        return
    
    # --- 메일 템플릿 읽기 ---
    with open(EMAIL_TEMPLATE_FILE, "r", encoding="utf-8") as f:
        template = f.read()
    
    # --- SMTP 설정 (테스트용) ---
    smtp_server = "smtp.example.com"
    port = 587
    username = "business@dadajikgu.com"
    password = "***SET***"
    sender_email = "business@dadajikgu.com"
    
    # --- 이메일 발송 ---
    sent_count = 0
    
    for client in clients[:3]:
        try:
            msg = MIMEMultipart("alternative")
            msg["Subject"] = "[주요] 중국 구매셀러 전용 — 통관 비용 절감 20%, 배송 시간 단축 1.5 일 보장"
            
            # 본문 내용 (HTML/텍스트)
            html_content = f"""
<html>
<body style="font-family: Arial; background-color: #F7F9FB;">
    <div style="border-left: 4px solid {#FF6B35}; padding: 20px;">
        <p style="color: {#004E89};" font-size="16px">안녕하세요, 다다직구 비즈니스 파트너십 담당자입니다.</p>
    </div>
    
    <h2 style="color: {#FF6B35}; margin-top: 30px;">통관 비용 절감 20% 및 배송 시간 단축 1.5 일 보장</h2>
    
    <table style="width: 80%; margin: auto;">
        <tr><td align="left" style="padding: 10px;"><strong>고객사명:</strong></td><td>{client["name"]}</td></tr>
        <tr><td align="left" style="padding: 10px;"><strong>연간 절감액 예상:</strong></td><td>{client.get("estimated_savings", "N/A")}</td></tr>

    <p style="margin-top: 20px;">[다운로드 버튼](#) → [무료 가이드 다운로드]</p>
</body>
</html>
            """
            
            part1 = MIMEText(html_content, "html")
            msg.attach(part1)
            
            # --- SMTP 서버 연결 및 발송 ---
            server = smtplib.SMTP(smtp_server, port)
            server.starttls()
            server.login(username, password)
            
            server.sendmail(sender_email, client["contact_email"], msg.as_string())
            server.quit()

            print(f"[완성] {client['name']} 이메일 발송 완료")
        
        except Exception as e:
            print(f"[오류] {client.get('name', '알 수 없음')} 이메일 발송 실패: {str(e)}")
    
    # --- 결과 저장 및 보고서 출력 ---
    with open("e:\work_company\agents\business\tools\email_send_report.json", "w", encoding="utf-8") as f:
        json.dump({"sent_count": sent_count, "timestamp": datetime.now()}, f)
    
    print(f"[완료] 총 {sent_count} 곳의 이메일 발송 완료 (2026-06-06)")

if __name__ == "__main__":
    send_bulk_emails()
